# With "everyone can use" off, only the members, DJs and admins are answered.
s everyoneCanUse=false
c !play test tone
e You are not allowed to control me.
s members=[{"uid":"$ME","name":"Tester"}] commandLevels={"pause":"dj"}
c !play test tone
e Now playing
c !pause
e [b]!pause[/b] is only for DJs and admins.
s members=[] djs=[{"uid":"$ME","name":"Tester"}]
c !pause
e Paused.
# A server group works like a person.
s djs=[] djGroups=[$GROUP]
c !np
e Now playing
# Command levels: !vol only for admins.
s commandLevels={"vol":"admin"}
c !vol 30
e [b]!vol[/b] is only for admins.
s admins=[{"uid":"$ME"}]
c !vol 30
e Volume set to 30%
# Blocked beats every group and list: no answer at all.
s blocked=[{"uid":"$ME"}]
c !np
x
p !xyz
