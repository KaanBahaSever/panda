# The equalizer from the chat.
s announce=false
c !eq
e Equalizer: [b]off[/b]
c !eq bass
e Equalizer: [b]bass[/b].
c !eq nothing
e There is no equalizer called
c !eq custom
e Set your own bands in the panel first
# Effects; !fx alone says what's on.
c !fx 8d
e 8D on
c !fx nightcore
e [b]nightcore[/b] on.
c !fx
e nightcore, 8D, bass
c !fx reverse
e Unknown effect
# Nightcore plays the song faster: 1.25 song seconds in a second.
c !play test tone
e Now playing
w 8
t 10 14
c !fx nightcore
e [b]nightcore[/b] off.
# The speed, without the pitch; turning it on goes on where the song was.
c !stop
c !speed 2
e Speed [b]×2[/b].
c !play test tone
w 5
t 11 17
c !speed 3
e The speed goes from 0.5 to 2
c !fx off
e Effects off, normal speed.
c !eq off
e Equalizer off.
